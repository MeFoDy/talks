const updateValidationStateForInput = (input) => {
    const isValid = input.checkValidity();
    const errorEl = input.nextElementSibling;

    errorEl.textContent = input.validationMessage;
    errorEl.hidden = isValid;

    if (isValid) {
        input.removeAttribute("aria-invalid");
    } else {
        input.setAttribute("aria-invalid", "true");
    }
};

Лучшие практики

  • Уникальные id и name для каждого поля.
  • Все input находятся внутри form.
  • Связанные элементы (например, логин и пароль) — в одной форме.
  • Никак не связанные элементы (например, пароль и дата рождения) — в разных формах.
  • Правильный method у форм.
  • Настроенный autocomplete у элементов.

function PhoneInput() {
    const [value, setValue] = useState('');
    const ref = useRef();

    const handleChange = (e) => {
        const cursorPosition = e.target.selectionStart;
        const formatted = formatPhone(e.target.value);
        setValue(formatted);

        requestAnimationFrame(() => {
            ref.current.setSelectionRange(cursorPosition, cursorPosition);
        });
    };

    return <input ref={ref} value={value} onChange={handleChange} />;
}
